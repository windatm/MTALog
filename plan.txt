class MTALog(nn.Module):
def init(self, encoder, classifier, ...):
super().init()
self.encoder = encoder # AttGRUModel hoặc tương đương
self.classifier = classifier

def encode_batch(self, x: torch.Tensor) -> torch.Tensor:
    """
    x: [N, seq_len, ...]
    return: z: [N, d]
    """
    z = self.encoder(x)
    return z


2) Hàm compute_prototypes(domain_batches, model):

from typing import Dict
from collections import defaultdict
import torch

def compute_prototypes(
domain_batches: List[DomainBatch],
model: MTALog,
) -> Dict[str, Dict[int, torch.Tensor]]:
"""
Trả về:
prototypes[domain_id][class_y] = μ_d^y (tensor [d])
"""
prototypes: Dict[str, Dict[int, torch.Tensor]] = {}

for db in domain_batches:
    z_supp = model.encode_batch(db.support_x)  # [N_supp, d]
    prot_per_class: Dict[int, torch.Tensor] = {}

    for y_val in (0, 1):
        mask = (db.support_y == y_val)
        if mask.sum() == 0:
            continue
        mu = z_supp[mask].mean(dim=0)
        prot_per_class[int(y_val)] = mu

    prototypes[db.domain_id] = prot_per_class

return prototypes

PHASE 3 – SOURCE RELIABILITY r_s VÀ DISTANCE D_s

Ý tưởng: với mỗi source s, dùng prototype của s để phân loại support target theo nearest-prototype → đo accuracy + margin (r_s). Đồng thời đo khoảng cách prototype source–target D_s. Kết hợp thành score_s.

Tạo file source_weighting.py.

from typing import Dict, List
import torch
from torch import nn

def compute_source_scores(
prototypes: Dict[str, Dict[int, torch.Tensor]],
domain_batches: List[DomainBatch],
model: MTALog,
w_r: float = 1.0,
w_D: float = 1.0,
beta_margin: float = 0.1,
) -> Dict[str, float]:
"""
Tính score_s cho từng source domain (không gồm 'TARGET').
"""
# 1. Lấy DomainBatch của TARGET
target_db = None
for db in domain_batches:
if db.domain_id == "TARGET":
target_db = db
break
if target_db is None:
raise ValueError("TARGET domain not found in domain_batches.")

z_T_supp = model.encode_batch(target_db.support_x)  # [N_T, d]
y_T_supp = target_db.support_y                     # [N_T]

scores: Dict[str, float] = {}

# 2. Lặp qua từng source domain
for db in domain_batches:
    if db.domain_id == "TARGET":
        continue
    s = db.domain_id

    prot_s = prototypes.get(s, {})
    prot_T = prototypes.get("TARGET", {})

    # 2.1. Tính D_s (khoảng cách prototype)
    D_s = 0.0
    for y_val in (0, 1):
        if y_val in prot_s and y_val in prot_T:
            D_s += torch.dist(prot_s[y_val], prot_T[y_val], p=2).item() ** 2

    # 2.2. Nearest-prototype prediction trên support TARGET
    preds = []
    margins = []

    for i in range(z_T_supp.size(0)):
        z_i = z_T_supp[i]

        # Giả sử cả 0 và 1 đều tồn tại; nếu thiếu class nào, cần xử lý edge case.
        d0 = torch.dist(z_i, prot_s[0], p=2) if 0 in prot_s else None
        d1 = torch.dist(z_i, prot_s[1], p=2) if 1 in prot_s else None

        if d0 is None or d1 is None:
            # Nếu thiếu prototype 0 hoặc 1, có thể bỏ qua hoặc đặt penalty lớn.
            # Ở đây tạm penalty, coi source này kém.
            preds.append(1 - int(y_T_supp[i].item()))  # cố ý sai
            margins.append(0.0)
            continue

        # true / wrong distance
        true_y = int(y_T_supp[i].item())
        d_true = d0 if true_y == 0 else d1
        d_wrong = d1 if true_y == 0 else d0

        pred_y = 0 if d0 < d1 else 1
        preds.append(pred_y)

        margin = (d_wrong - d_true).item()
        margins.append(margin)

    if len(preds) == 0:
        # không đủ thông tin, score rất thấp
        scores[s] = -1e6
        continue

    preds_t = torch.tensor(preds, device=y_T_supp.device)
    acc_s = (preds_t == y_T_supp).float().mean().item()
    margin_s = float(sum(margins) / len(margins))

    r_s = acc_s + beta_margin * margin_s

    # 2.3. Kết hợp r_s và D_s
    score_s = w_r * r_s - w_D * D_s
    scores[s] = score_s

return scores

PHASE 4 – SCORES → α_s, TÍCH HỢP VÀO LOSS

Hàm chuyển score_s → α_s (softmax):

def scores_to_alphas(scores: Dict[str, float], kappa: float = 1.0) -> Dict[str, float]:
import torch
keys = list(scores.keys())
vals = torch.tensor([scores[k] for k in keys])
alphas_tensor = torch.softmax(kappa * vals, dim=0)
return {k: float(v) for k, v in zip(keys, alphas_tensor)}

def train_on_episode(self, domain_batches: List[DomainBatch]) -> dict:
self.train()
# 1. Compute prototypes
prototypes = compute_prototypes(domain_batches, self)

# 2. Compute source scores & alphas
scores = compute_source_scores(
    prototypes=prototypes,
    domain_batches=domain_batches,
    model=self,
    w_r=1.0,
    w_D=1.0,
    beta_margin=0.1,
)
alphas = scores_to_alphas(scores, kappa=1.0)
# alphas: {'BGL': α_BGL, 'HDFS': α_HDFS, ...}

# 3. Compute L_cls (weighted CE trên query các domain)
L_cls = 0.0
total_weight = 0.0

for db in domain_batches:
    z_query = self.encode_batch(db.query_x)          # [N_q, d]
    logits = self.classifier(z_query)                # [N_q, 2]
    ce = F.cross_entropy(logits, db.query_y, reduction="none")  # [N_q]

    if db.domain_id == "TARGET":
        w_domain = 1.0  # hoặc β cố định
    else:
        w_domain = alphas.get(db.domain_id, 0.0)

    L_cls = L_cls + w_domain * ce.sum()
    total_weight += w_domain * ce.numel()

if total_weight > 0:
    L_cls = L_cls / total_weight
else:
    L_cls = torch.tensor(0.0, device=next(self.parameters()).device)

# 4. Compute L_align (ví dụ dùng prototype normal)
# Ví dụ đơn giản: align prototype normal target với weighted average normal sources

lambda_align = self.lambda_align
lambda_cl = self.lambda_cl

if 0 in prototypes.get("TARGET", {}):
    mu_T_0 = prototypes["TARGET"][0]
    src_mu0_weighted = None
    norm_alpha = 0.0

    for db in domain_batches:
        if db.domain_id == "TARGET":
            continue
        s = db.domain_id
        if 0 not in prototypes[s]:
            continue
        alpha_s = alphas.get(s, 0.0)
        if src_mu0_weighted is None:
            src_mu0_weighted = alpha_s * prototypes[s][0]
        else:
            src_mu0_weighted = src_mu0_weighted + alpha_s * prototypes[s][0]
        norm_alpha += alpha_s

    if src_mu0_weighted is not None and norm_alpha > 0:
        src_mu0_weighted = src_mu0_weighted / norm_alpha
        L_align = torch.dist(mu_T_0, src_mu0_weighted, p=2) ** 2
    else:
        L_align = torch.tensor(0.0, device=mu_T_0.device)
else:
    L_align = torch.tensor(0.0, device=next(self.parameters()).device)

# 5. Compute L_CL (nếu đã có hàm supcon_loss)
# Ví dụ: gom tất cả embedding (support + query), gán weight theo alpha_s / 1.0

# pseudo:
# z_all, y_all, w_all = [], [], []
# for db in domain_batches:
#     z_supp = self.encode_batch(db.support_x)
#     z_query = self.encode_batch(db.query_x)
#     if db.domain_id == "TARGET":
#         w_domain = 1.0
#     else:
#         w_domain = alphas.get(db.domain_id, 0.0)
#     append z_supp, z_query vào z_all; y vào y_all; weight w_domain vào w_all
#
# L_CL = supcon_loss(z_all, y_all, weights=w_all)

L_CL = torch.tensor(0.0, device=next(self.parameters()).device)  # placeholder

L_total = L_cls + lambda_align * L_align + lambda_cl * L_CL

self.optimizer.zero_grad()
L_total.backward()
self.optimizer.step()

return {
    "L_total": float(L_total.item()),
    "L_cls": float(L_cls.item()),
    "L_align": float(L_align.item()),
    "L_CL": float(L_CL.item()),
}

PHASE 6 – THÍ NGHIỆM & ABLATION

Thêm flag cấu hình: source_weighting_mode ∈ {"none", "distance_only", "reliability_plus_distance"}.

Trong compute_source_scores():

Nếu mode = "none": luôn trả scores = {s: 0.0}, sau đó alphas = uniform.

Nếu mode = "distance_only":
score_s = -D_s

Nếu mode = "reliability_plus_distance":
score_s = w_r * r_s - w_D * D_s

Thực nghiệm:

Mode "none": baseline meta (multi-source, không weighting).

Mode "distance_only": chỉ dựa trên khoảng cách prototype.

Mode "reliability_plus_distance": full cơ chế đề xuất.

So sánh F1/AUC trên target.
